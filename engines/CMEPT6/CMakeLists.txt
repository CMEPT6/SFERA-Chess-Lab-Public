cmake_minimum_required(VERSION 3.16)
project(CMEPT6_Engine VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cmept6-engine src/main.cpp)
if(MSVC)
    # Portable MSVC runtime: users do not need a separate VC++ redistributable.
    set_property(TARGET cmept6-engine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(cmept6-engine PRIVATE /W4 /O2)
else()
    target_compile_options(cmept6-engine PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()

option(CMEPT6_CHECK_INVARIANTS "Verify board invariants during perft (slow)" OFF)
if(CMEPT6_CHECK_INVARIANTS)
    target_compile_definitions(cmept6-engine PRIVATE CMEPT6_CHECK_INVARIANTS=1)
endif()

option(CMEPT6_BUILD_INTERNAL_TESTS "Build extra correctness tests" OFF)
if(CMEPT6_BUILD_INTERNAL_TESTS)
    add_executable(cmept6-slider-fuzz tests/slider_fuzz.cpp)
    if(MSVC)
        target_compile_options(cmept6-slider-fuzz PRIVATE /W4 /O2)
    else()
        target_compile_options(cmept6-slider-fuzz PRIVATE -O2 -Wall -Wextra -Wpedantic)
    endif()
endif()
